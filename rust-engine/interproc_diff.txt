+

+        if chars[i..i + r_chars.len()] == r_chars {

+fn collect_method_instructions(
+    body: &[ir::InstructionId],
+    program: &ir::Program,
+    out: &mut Vec<ir::InstructionId>,
+    visited: &mut HashSet<ir::InstructionId>,

+                    collect_method_instructions(then_block, program, out, visited);
+                    if let Some(eb) = else_block {
+                        collect_method_instructions(eb, program, out, visited);
+                    }
+                }
+                InstructionKind::Loop { body, .. } => {
+                    collect_method_instructions(body, program, out, visited);
+                }
+                InstructionKind::Try { body, catches, finally, .. } => {
+                    collect_method_instructions(body, program, out, visited);
+                    for catch_id in catches {
+                        collect_method_instructions(&[*catch_id], program, out, visited);
+                    }
+                    if let Some(fb) = finally {
+                        collect_method_instructions(fb, program, out, visited);
+                    }
+                }
+                InstructionKind::Catch { body, .. } => {
+                    collect_method_instructions(body, program, out, visited);
+                }
+                _ => {}
+            }
+        }
+    }
